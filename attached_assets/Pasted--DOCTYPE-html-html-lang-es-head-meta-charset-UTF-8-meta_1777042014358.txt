let cart = JSON.parse(localStorage.getItem('domo3d_cart') || '[]');
const saveCart = () => localStorage.setItem('domo3d_cart', JSON.stringify(cart));

function cartTotals(){
  const sub = cart.reduce((s,i) => s + i.price*i.qty, 0);
  const iva = Math.round(sub*.19);
  const tot = sub + iva;
  return { sub, iva, tot, half: Math.round(tot*.5) };
}

function updateBadge(){
  const b = document.getElementById('cbadge');
  const n = cart.reduce((s,i) => s+i.qty, 0);
  if(n > 0){ b.textContent=n; b.style.display='flex'; }
  else b.style.display='none';
}

function addToCart(id){
  const p = PRODUCTS.find(x => x.id===id);
  const ex = cart.find(x => x.id===id);
  ex ? ex.qty++ : cart.push({id:p.id,name:p.name,price:p.price,qty:1});
  saveCart(); updateBadge(); renderCart();
  const b = document.getElementById('cbadge');
  b.classList.add('bump'); setTimeout(()=>b.classList.remove('bump'),300);
  showToast(`<i class="fa-solid fa-circle-check" style="color:var(--c);margin-right:.4rem"></i>${p.name} agregado al carrito`);
}

function removeFromCart(id){ cart=cart.filter(x=>x.id!==id); saveCart(); updateBadge(); renderCart(); }
function changeQty(id,d){ const i=cart.find(x=>x.id===id); if(i){i.qty=Math.max(1,i.qty+d); saveCart(); updateBadge(); renderCart();} }

function renderCart(){
  const el = document.getElementById('cdr-items');
  const em = document.getElementById('cdr-empty');
  const ft = document.getElementById('cdr-foot');
  el.innerHTML='';
  if(!cart.length){ el.style.display='none'; em.style.display='flex'; ft.style.display='none'; return; }
  el.style.display='block'; em.style.display='none'; ft.style.display='block';
  cart.forEach(item => {
    const p = PRODUCTS.find(x=>x.id===item.id);
    const d = document.createElement('div'); d.className='ci';
    d.innerHTML=`
      <div class="ci-ic"><i class="fa-solid ${p.icon}" style="font-size:.8rem;color:${p.col}"></i></div>
      <div class="ci-info">
        <p>${item.name}</p>
        <small style="color:${p.col}">${fmtCOP(item.price)}</small>
      </div>
      <div class="ci-ctrl">
        <button onclick="changeQty(${item.id},-1)"><i class="fa-solid fa-minus"></i></button>
        <span>${item.qty}</span>
        <button onclick="changeQty(${item.id},1)"><i class="fa-solid fa-plus"></i></button>
      </div>
      <button class="ci-del" onclick="removeFromCart(${item.id})"><i class="fa-solid fa-trash"></i></button>`;
    el.appendChild(d);
  });
  const {sub,iva,tot,half}=cartTotals();
  document.getElementById('cs').textContent=fmtCOP(sub);
  document.getElementById('ci2').textContent=fmtCOP(iva);
  document.getElementById('ct').textContent=fmtCOP(tot);
  document.getElementById('ch').textContent=fmtCOP(half);
}

   KEYFRAMES DE CÁMARA
   Cada frame: t (0-1), pos [x,y,z], look [x,y,z]
───────────────────────────────────────────────────── */
const KF = [
  // Acto I: Vista cenital, orbita suave
  { t:0.00, p:[ 0,  22,  6], l:[0,0, 0] },
  { t:0.08, p:[ 7,  17, 14], l:[0,1, 0] },
  { t:0.15, p:[ 2,  14, 16], l:[0,2, 0] },
  // Acto II: Descenso hacia puerta
  { t:0.20, p:[ 1,   9, 17], l:[0,2.5,0] },
  { t:0.27, p:[ 0,   5, 11], l:[0,3,  0] },
  { t:0.34, p:[-1, 3.8,  7], l:[0,3.5,0] },
  // Acto III: 4 puntos de interés internos
  // 3a: Bombilla techo (centro sala)
  { t:0.40, p:[ 0, 4.5,  2], l:[ 0, 5,   -0.5] },
  // 3b: Interruptor pared derecha
  { t:0.51, p:[ 3, 3.2,  0.5], l:[5.2, 3,  0  ] },
  // 3c: Cámara pared izquierda
  { t:0.61, p:[-2.5, 4.2, -0.5], l:[-5.5, 3.8, -1] },
  // 3d: Contenedor suelo
  { t:0.70, p:[ 1.5, 2.5,  4], l:[3.5, 0.8, 3.5] },
  // Acto IV: Plano general iluminado
  { t:0.77, p:[-11, 12, 20], l:[0,2,0] },
  { t:0.88, p:[  0,  9, 23], l:[0,2,0] },
  // Acto V: Pantalla de video (frente a la casa)
  { t:0.93, p:[  0,  7, 21], l:[0,5,12] },
  { t:1.00, p:[  0,  5, 19], l:[0,4.5,12] },
];

function smoothstep(x){ return x*x*(3-2*x) }

function interpCam(t){
  let i=0;
  while(i < KF.length-2 && KF[i+1].t <= t) i++;
  const a=KF[i], b=KF[i+1];
  const span = b.t - a.t;
  const s = smoothstep(span>0 ? Math.max(0,Math.min(1,(t-a.t)/span)) : 0);
  return {
    x: a.p[0]+(b.p[0]-a.p[0])*s, y: a.p[1]+(b.p[1]-a.p[1])*s, z: a.p[2]+(b.p[2]-a.p[2])*s,
    lx:a.l[0]+(b.l[0]-a.l[0])*s, ly:a.l[1]+(b.l[1]-a.l[1])*s, lz:a.l[2]+(b.l[2]-a.l[2])*s,
  };
}

/* ─────────────────────────────────────────────────────
   INIT THREE
───────────────────────────────────────────────────── */
function initThree(){
  const canvas = document.getElementById('stage');
  renderer = new THREE.WebGLRenderer({ canvas, antialias:true });
  renderer.setPixelRatio(Math.min(window.devicePixelRatio,2));
  renderer.setSize(window.innerWidth, window.innerHeight);
  renderer.shadowMap.enabled = true;
  renderer.shadowMap.type = THREE.PCFSoftShadowMap;

  scene = new THREE.Scene();
  scene.background = new THREE.Color(0x040c0d);
  scene.fog = new THREE.FogExp2(0x040c0d, 0.016);

  camera = new THREE.PerspectiveCamera(52, window.innerWidth/window.innerHeight, 0.1, 200);
  camera.position.set(0,22,6);

  clock = new THREE.Clock();

  window.addEventListener('resize',()=>{
    camera.aspect = window.innerWidth/window.innerHeight;
    camera.updateProjectionMatrix();
    renderer.setSize(window.innerWidth, window.innerHeight);
  });

  buildLights();
  buildGround();
  buildHouse();
  buildDevices();
  buildNetwork();
  buildCity();
  buildVideoScreen();
  buildPeople();
  startRenderLoop();
}

function addEdges(mesh, color=0x00F0FF, opacity=0.18){
  const m=new THREE.LineBasicMaterial({color,transparent:true,opacity});
  const e=new THREE.LineSegments(new THREE.EdgesGeometry(mesh.geometry),m);
  mesh.add(e);
}

/* ─────────────────────────────────────────────────────
   RED NEURONAL (partículas + líneas)
───────────────────────────────────────────────────── */
function buildNetwork(){
  const N=90; const nodes=[];
  for(let i=0;i<N;i++){
    const th=Math.random()*Math.PI*2;
    const ph=Math.acos(2*Math.random()-1);
    const r=14+Math.random()*22;
    nodes.push(new THREE.Vector3(
      r*Math.sin(ph)*Math.cos(th),
      r*Math.sin(ph)*Math.sin(ph)*0.55+4,
      r*Math.cos(ph)
    ));
  }
  // Points
  const pgeo=new THREE.BufferGeometry();
  pgeo.setAttribute('position',new THREE.Float32BufferAttribute(nodes.flatMap(n=>[n.x,n.y,n.z]),3));
  const pmat=new THREE.PointsMaterial({color:0x00F0FF,size:.2,transparent:true,opacity:.5});
  tracked.netPtMat=pmat;
  const pts=new THREE.Points(pgeo,pmat);
  scene.add(pts); window._netPts=pts;

  // Lines
  const lv=[];
  for(let i=0;i<nodes.length;i++)
    for(let j=i+1;j<nodes.length;j++)
      if(nodes[i].distanceTo(nodes[j])<9){ lv.push(nodes[i].x,nodes[i].y,nodes[i].z,nodes[j].x,nodes[j].y,nodes[j].z); }
  const lgeo=new THREE.BufferGeometry();
  lgeo.setAttribute('position',new THREE.Float32BufferAttribute(lv,3));
  const lmat=new THREE.LineBasicMaterial({color:0x00F0FF,transparent:true,opacity:.07});
  tracked.netLineMat=lmat;
  scene.add(new THREE.LineSegments(lgeo,lmat));
}

/* ─────────────────────────────────────────────────────
   RENDER LOOP
───────────────────────────────────────────────────── */
let scrollProg = 0;   // actualizado por ScrollTrigger

function startRenderLoop(){
  function tick(){
    requestAnimationFrame(tick);
    const t = clock.getElapsedTime();

    // Interpolar cámara
    const kf = interpCam(scrollProg);
    camTarget.x=kf.x; camTarget.y=kf.y; camTarget.z=kf.z;
    camTarget.lx=kf.lx; camTarget.ly=kf.ly; camTarget.lz=kf.lz;

    const ease=0.055;
    for(const k of ['x','y','z','lx','ly','lz'])
      camCurrent[k]+=(camTarget[k]-camCurrent[k])*ease;

    camera.position.set(camCurrent.x,camCurrent.y,camCurrent.z);
    camera.lookAt(camCurrent.lx,camCurrent.ly,camCurrent.lz);

    // Luz fill sigue cámara
    if(window._fillLight)
      window._fillLight.position.lerp(new THREE.Vector3(camCurrent.x,camCurrent.y-2,camCurrent.z),.06);

    // Red neuronal: pulso
    const pulse = Math.sin(t*.75)*.055+.09;
    if(tracked.netLineMat) tracked.netLineMat.opacity=pulse;
    if(tracked.netPtMat)   tracked.netPtMat.opacity=.4+Math.sin(t*.5)*.1;
    if(window._netPts)     window._netPts.rotation.y+=.00035;

    // Activar dispositivos
    updateDevices(scrollProg, t);

    // Personajes bobbing
    tracked.personMeshes.forEach(p=>{
      const sh=Math.max(0,Math.min(1,(scrollProg-.72)/.08));
      p.mesh.visible=sh>.01;
      p.mesh.position.y=p.baseY+Math.sin(t*1.3+p.phase)*.055;
    });

    renderer.render(scene,camera);
  }
  tick();
}

/* ─────────────────────────────────────────────────────
   ACTIVACIÓN DE DISPOSITIVOS
───────────────────────────────────────────────────── */
function bell(t,s,e){ if(t<s||t>e)return 0; return Math.sin(((t-s)/(e-s))*Math.PI); }
function ramp(t,s,e){ return Math.max(0,Math.min(1,(t-s)/(e-s))); }

function updateDevices(t, time){
  const global = ramp(t,.32,.5);
  const pulse  = 1+Math.sin(time*2.8)*.14;

  // Bombilla
  const ba = Math.max(bell(t,.36,.52), global*.3);
  if(tracked.bulbMat){
    tracked.bulbMat.emissiveIntensity = .12 + ba*2.2*pulse;
    tracked.bulbMat.emissive.setHex(lerp32(0x220800,0xFFAA00,Math.min(1,ba*1.6)));
  }
  if(tracked.bulbLight) tracked.bulbLight.intensity = ba*2.8;

  // Interruptor
  const sa = Math.max(bell(t,.48,.63), global*.25);
  if(tracked.swMat){
    tracked.swMat.emissiveIntensity = .1 + sa*1.8*pulse;
    tracked.swMat.emissive.setHex(lerp32(0x120040,0x5500CC,Math.min(1,sa*1.5)));
  }
  if(tracked.swLight) tracked.swLight.intensity = sa*1.5;

  // Cámara
  const ca = Math.max(bell(t,.58,.73), global*.2);
  if(tracked.camMat){
    tracked.camMat.emissiveIntensity = .1 + ca*1.8*pulse;
    tracked.camMat.emissive.setHex(lerp32(0x200008,0xAA0000,Math.min(1,ca*1.5)));
  }
  if(tracked.camLedMat) tracked.camLedMat.emissiveIntensity = .4+ca*1.4;

  // Contenedor
  const ta = Math.max(bell(t,.66,.78), global*.2);
  if(tracked.trashMat){
    tracked.trashMat.emissiveIntensity = .1 + ta*1.8*pulse;
    tracked.trashMat.emissive.setHex(lerp32(0x001208,0x00664A,Math.min(1,ta*1.5)));
  }
  if(tracked.trashLight) tracked.trashLight.intensity = ta*1.6;

  // Paredes
  const wallGlow = global*.28;
  tracked.houseWallMats.forEach(m=>{ m.emissiveIntensity = .1+wallGlow; });
  if(tracked.roofMat) tracked.roofMat.emissiveIntensity = .15+wallGlow*.8;

  // Ventanas
  const winBright = ramp(t,.3,.65);
  tracked.winMats.forEach(m=>{
    m.emissiveIntensity = .05+winBright*.9;
    m.emissive.setHex(lerp32(0x000d16,0x003344,winBright));
  });

  // Interior light
  if(tracked.interiorLight) tracked.interiorLight.intensity = ramp(t,.3,.55)*1.6;

  // Pantalla video (Acto V)
  const va = ramp(t,.87,.96);
  if(tracked.screenMat) tracked.screenMat.emissiveIntensity = .05+va*.8;
  if(tracked.screenLight) tracked.screenLight.intensity = va*2.5;
}

/* Helper: interpola dos colores hex */
function lerp32(a,b,t){
  const ar=(a>>16)&0xFF, ag=(a>>8)&0xFF, ab=a&0xFF;
  const br=(b>>16)&0xFF, bg=(b>>8)&0xFF, bb=b&0xFF;
  const r=Math.round(ar+(br-ar)*t);
  const g=Math.round(ag+(bg-ag)*t);
  const bl2=Math.round(ab+(bb-ab)*t);
  return (r<<16)|(g<<8)|bl2;
}

/* ─────────────────────────────────────────────────────
   GSAP SCROLL TRIGGER
───────────────────────────────────────────────────── */
function initScroll(){
  gsap.registerPlugin(ScrollTrigger);

  ScrollTrigger.create({
    trigger:'#scroll-driver',
    start:'top top',
    end:'bottom bottom',
    scrub:1.4,
    onUpdate(self){
      const p = self.progress;
      scrollProg = p;
      document.getElementById('pbar').style.width = (p*100)+'%';
      updateOverlays(p);
      updateKwh(p);
    }
  });

  // Fade del canvas al llegar al post-scroll
  ScrollTrigger.create({
    trigger:'#post',
    start:'top 96%',
    end:'top 30%',
    scrub:true,
    onUpdate(self){
      const op = Math.max(0, 1-self.progress*1.3);
      document.getElementById('stage').style.opacity=op;
      if(self.progress>.6) hideAllOverlays();
    }
  });
}

/* ─────────────────────────────────────────────────────
   CONTROL DE OVERLAYS
───────────────────────────────────────────────────── */
function setOv(id, t, start, end, fade=0.032){
  const el=document.getElementById(id); if(!el) return;
  let op=0;
  if(t<start) op=0;
  else if(t<start+fade) op=(t-start)/fade;
  else if(t<end-fade) op=1;
  else if(t<end) op=(end-t)/fade;
  else op=0;
  el.style.opacity=op;
  el.style.pointerEvents=op>.06 ? 'auto' : 'none';
}

function setPCard(id, t, start, end){
  const el=document.getElementById(id); if(!el) return;
  const fade=0.022;
  let op=0;
  if(t>=start && t<start+fade) op=(t-start)/fade;
  else if(t>=start+fade && t<end-fade) op=1;
  else if(t>=end-fade && t<end) op=(end-t)/fade;
  el.style.opacity=op;
  el.style.pointerEvents=op>.06 ? 'auto' : 'none';
}

function updateOverlays(t){
  setOv('a1', t, 0,    0.175);
  setOv('a2', t, 0.15, 0.375);
  setOv('a3', t, 0.34, 0.78);
  setOv('a4', t, 0.74, 0.925);
  setOv('a5', t, 0.90, 0.995);
  // Cards de producto individuales
  setPCard('p1', t, 0.37, 0.51);
  setPCard('p2', t, 0.49, 0.63);
  setPCard('p3', t, 0.59, 0.73);
  setPCard('p4', t, 0.68, 0.79);
}

function hideAllOverlays(){
  ['a1','a2','a3','a4','a5'].forEach(id=>{
    const el=document.getElementById(id);
    if(el){ el.style.opacity=0; el.style.pointerEvents='none'; }
  });
}

function updateKwh(t){
  const saved=Math.round(105000*Math.min(1,t*1.4));
  const kwh=Math.round(105*Math.min(1,t*1.4));
  const el=document.getElementById('kwh-val');
  if(el) el.textContent = saved>0
    ? `-${kwh} kWh · -${(saved/1000).toFixed(0)}K COP/mes`
    : '0 kWh ahorrados';
}
